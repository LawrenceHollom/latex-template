%% General semantics

% Defined :=
\newcommand{\defined}{\mathrel{\coloneqq}}

% Parentheses (...)
\DeclarePairedDelimiter{\p}{\lparen}{\rparen}
\let\b\relax
\DeclarePairedDelimiter{\b}{\lbrack}{\rbrack}

% Such that
\newcommand{\st}{\mathbin{\colon}}

% Set
\undef{\set}
\DeclarePairedDelimiter{\set}{\lbrace}{\rbrace}

% Empty set
\undef{\emptyset}
\newcommand{\emptyset}{\varnothing}

% Cardinality
\DeclarePairedDelimiter{\card}{\lvert}{\rvert}

% Union and complement
\newcommand{\union}{\mathbin{\cup}}
\newcommand{\inter}{\mathbin{\cap}}
\newcommand{\bigunion}{\mathbin{\bigcup}}
\newcommand{\biginter}{\mathbin{\bigcap}}

% Indicator function
\DeclareMathOperator{\ind}{\mathbf{1}}

% From (i.e. $f \from X \to Y$ gives "f : X -> Y")
\newcommand{\from}{\colon}

% Image
\newcommand{\im}{\text{im}}

% Floor and ceil
\DeclarePairedDelimiter{\floor}{\lfloor}{\rfloor}
\DeclarePairedDelimiter{\ceil}{\lceil}{\rceil}

% Expectation, e.g. $\expects_{x \in G} f(x)$
\DeclareMathOperator*{\expecs}{\scalerel*{\mathbb{E}}{A_B}}

% Setminus a singleton
\newcommand{\setm}[1]{\setminus\set{#1}}

% Better differentials.
\let\d\relax
\newcommand{\d}{\, \mathrm{d}}

%----------------------------------------------------------%

%% Number theory

% Hungarian chickens
% (allegedly Erdos once claimed that the chickens in Hungary should be taught to say "log log log".)
\DeclareMathOperator{\loglog}{log\,log}
\DeclareMathOperator{\logloglog}{log\,log\,log}
\DeclareMathOperator{\loglogloglog}{log\,log\,log\,log}

% Divisibility
\newcommand{\divides}{\mathrel{\mid}}
\DeclareMathOperator{\eqmod}{\equiv}

% Modular arithmetic
\undef{\mod}
\newcommand{\mod}[1]{\ (\mathrm{mod}\ #1)}

%----------------------------------------------------------%

%% Vector spaces etc.

% Absolute value
\undef{\abs}
\DeclarePairedDelimiterX{\abs}[1]
  {\lvert}{\rvert}{\ifblank{#1}{\,\cdot\,}{#1}}

% Norm
\undef{\norm}
\DeclarePairedDelimiterX{\norm}[1]
  {\lVert}{\rVert}{\ifblank{#1}{\,\cdot\,}{#1}}

% Inner Product
\DeclarePairedDelimiterX{\inner}[2]
  {\langle}{\rangle}{\ifblank{#1}{\,\cdot\,}{#1},\ifblank{#2}{\,\cdot\,}{#2}}

\DeclarePairedDelimiterX{\absinner}[2]
  {|\langle}{\rangle|}{\ifblank{#1}{\,\cdot\,}{#1},\ifblank{#2}{\,\cdot\,}{#2}}

% Embedding hook
\DeclareMathOperator{\embeds}{\hookrightarrow}

%----------------------------------------------------------%

%% Probability

% Distributed as
\DeclareMathOperator{\dist}{\sim}

% With high probability
\newcommand{\whp}{w.h.p. }

% Independently identically distributed
\newcommand{\iid}{i.i.d. }

% Almost surely
\newcommand{\as}{a.s. }

% NOTE in the above three commands, latex might treat '.' as the end of a sentence

% Conditional Symbol
\DeclareMathDelimiter{\given}
  {\mathbin}{symbols}{"6A}{largesymbols}{"0C}

% NOTE that some of the below ops don't work with line breaks.
% If you want a line break, use e.g. \prob rather than \Prob

% Probability
\DeclareMathOperator{\Prob}{\mathbb{P}}
\DeclarePairedDelimiterXPP{\prob}[1]
  {\Prob}{\lparen}{\rparen}{}
  {\renewcommand{\given}{\nonscript\;\delimsize\vert\nonscript\;\mathopen{}}#1}

% Expectation
\DeclareMathOperator{\Expec}{\mathbb{E}}
\DeclarePairedDelimiterXPP{\expec}[1]
  {\Expec}{\lparen}{\rparen}{}
  {\renewcommand{\given}{\nonscript\;\delimsize\vert\nonscript\;\mathopen{}}#1}

% Variance
\DeclareMathOperator{\Var}{Var}
\DeclarePairedDelimiterXPP{\var}[1]
  {\Var}{\lparen}{\rparen}{}
  {\renewcommand{\given}{\nonscript\;\delimsize\vert\nonscript\;\mathopen{}}#1}

% Covariance (Binary only, for more than two variables, use \cov)
\DeclareMathOperator{\Cov}{Cov}
\DeclarePairedDelimiterXPP{\cov}[2]
  {\Cov}{\lparen}{\rparen}{}{#1,#2}
